BASELINE = Path(r'D:\ProgramFiles\phpstudy_pro\Extensions\php\php7.3.4nts')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def snapshot(path):
    return {str(p.relative_to(path)): sha(p) for p in path.rglob('*') if p.is_file()}


def fixture(label, collision=False):
    target = LAB / label / 'PHP 7.3.4 test'
    target.mkdir(parents=True)
    with zipfile.ZipFile(ARCHIVE) as z:
        for name in z.namelist():
            original = BASELINE / name
            if original.is_file() and original.suffix.lower() != '.ini':
                dest = target / name
                dest.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(original, dest)
    extension = 'php_mbstring.dll' if collision else 'php_redis.dll'
    shutil.copy2(BASELINE / 'ext/php_redis.dll', target / 'ext' / extension)
    (target / 'php.ini').write_text('extension_dir="' + str(target / 'ext') + '"\n'
                                   'extension=' + extension + '\n'
                                   'session.upload_progress.enabled=1\ndate.timezone=Asia/Shanghai\n', encoding='utf-8')
    (target / 'custom').mkdir()
    (target / 'custom/keep.txt').write_text('custom file retained', encoding='utf-8')
    return target


def ps_quote(value):
    return "'" + str(value).replace("'", "''") + "'"


def call(script, target, args=(), unknown=False):
    driver = LAB / ('driver-' + str(time.time_ns()) + '.ps1')
    if unknown:
        inventory = "@((New-Object PSObject -Property @{ Name='php-cgi.exe'; ProcessId=1; ExecutablePath=$null }))"
    else:
        inventory = ("@(Microsoft.PowerShell.Management\\Get-WmiObject -Class $Class -Filter $Filter | Where-Object { "
                     "$_.ExecutablePath -and $_.ExecutablePath.StartsWith(" + ps_quote(str(LAB) + '\\') + ", [StringComparison]::OrdinalIgnoreCase) })")
    text = ("$global:LASTEXITCODE=0\n"
            "function Get-WmiObject { [CmdletBinding()] param([string]$Class,[string]$Filter)\n" + inventory + "\n}\n"
            "& " + ps_quote(ROOT / script) + " -PhpDir " + ps_quote(target) + " " + " ".join(ps_quote(a) for a in args) + "\n"
            "exit $LASTEXITCODE\n")
    for name in ('-ArchivePath', '-RuntimeDir', '-CheckOnly'):
        text = text.replace(ps_quote(name), name)
    driver.write_text(text, encoding='utf-8-sig')
    result = subprocess.run(['powershell.exe', '-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', str(driver)],
                            capture_output=True, timeout=180, env=ENV)
    output = result.stdout + result.stderr
    driver.with_suffix('.output.txt').write_bytes(output)
    return result.returncode, output.decode('gb18030', 'replace')


def receipt(path):
    tree = ET.parse(path)
    mapping = {}
    for entry in tree.findall('.//{*}DCT/{*}En'):
        key = next((c for c in entry if c.attrib.get('N') == 'Key'), None)
        value = next((c for c in entry if c.attrib.get('N') == 'Value'), None)
        if key is not None and value is not None:
            mapping[key.text] = value.text
    return mapping


def check(label, condition, detail=None):
    RESULTS.append({'case': label, 'passed': bool(condition)})
    if not condition:
        raise AssertionError(label + ': ' + str(detail))

# An actual native directory handle reproduces transient rename denial. Release
# the handle after the child reports readiness, then require a successful move.
locked = LAB / 'transient-lock' / 'source'
locked.mkdir(parents=True)
(locked / 'keep.txt').write_text('directory-lock-fixture', encoding='ascii')
destination = locked.parent / 'renamed'
ready = locked.parent / 'ready.txt'
driver = locked.parent / 'move.ps1'
driver.write_text(". " + ps_quote(ROOT / 'Php734-Legacy.Common.ps1') + "\n"
                  "function Get-WmiObject { param($Class,$Filter) @() }\n"
                  "[IO.File]::WriteAllText(" + ps_quote(ready) + ", 'ready')\n"
                  "Move-LegacyDirectory " + ps_quote(locked) + ' ' + ps_quote(destination) + ' ' + ps_quote(locked) + "\n"
                  "Write-Host LEGACY-MOVE-OK\n", encoding='utf-8-sig')
kernel = ctypes.WinDLL('kernel32', use_last_error=True)
kernel.CreateFileW.argtypes = [ctypes.c_wchar_p, ctypes.c_uint32, ctypes.c_uint32, ctypes.c_void_p, ctypes.c_uint32, ctypes.c_uint32, ctypes.c_void_p]
kernel.CreateFileW.restype = ctypes.c_void_p
kernel.CloseHandle.argtypes = [ctypes.c_void_p]
handle = kernel.CreateFileW(str(locked), 0x80, 3, None, 3, 0x02000000, None)
assert handle not in (None, ctypes.c_void_p(-1).value)
released = threading.Event()
def release_handle():
    kernel.CloseHandle(handle)
    released.set()
child = subprocess.Popen(['powershell.exe', '-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', str(driver)],
                          stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=ENV)
timer = None
try:
    deadline = time.monotonic() + 15
    while not ready.exists() and time.monotonic() < deadline and child.poll() is None:
        time.sleep(0.03)
    assert ready.exists(), 'Move child did not become ready'
    timer = threading.Timer(0.8, release_handle)
    timer.start()
    out, err = child.communicate(timeout=20)
    check('directory swap tolerates a real transient native directory lock', child.returncode == 0 and
          b'LEGACY-MOVE-OK' in out and (destination / 'keep.txt').read_text() == 'directory-lock-fixture',
          (out + err).decode('gb18030', 'replace')[-1800:])
finally:
    if timer is not None:
        timer.join(timeout=3)
    if not released.is_set():
        release_handle()
    if child.poll() is None:
        child.terminate()
        child.wait(timeout=5)
